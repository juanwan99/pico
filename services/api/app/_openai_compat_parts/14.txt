                        status="cancelled",

                        task_id=task_id,
                    )
                )

                raise

                await _finalize_run(run_id, status="failed", error=str(e), task_id=task_id)
                finalized = True
            finally:

                    await asyncio.shield(
                        _finalize_run(
                            run_id,
                            status="cancelled",
                            error="stream disconnected",
                            task_id=task_id,
                        )
                    )
